name: CI Pipeline

on:
  push:
    branches: [ "main" ]
  pull_request:
    branches: [ "main" ]

jobs:
  build-test-sonarqube-ort-docker:
    runs-on: ubuntu-latest

    env:
      SONAR_HOST_URL: ${{ secrets.SONAR_HOST_URL }}
      SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}
      DOCKER_IMAGE: myapp:latest

    steps:

      - name: Checkout repository
        uses: actions/checkout@v4


      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'


      - name: Build project
        run: |
          ./gradlew clean build --no-daemon
        # For Maven: mvn clean install -B


      - name: Run unit tests
        run: |
          ./gradlew test --no-daemon
        # For Maven: mvn test -B

      - name: SonarQube Scan
        uses: sonarsource/sonarqube-scan-action@v2
        with:
          args: >
            -Dsonar.projectKey=my_project_key
            -Dsonar.projectName=My Project
            -Dsonar.sources=src
            -Dsonar.java.binaries=build
        env:
          SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}
          SONAR_HOST_URL: ${{ secrets.SONAR_HOST_URL }}

     
      - name: Install ORT
        run: |
          curl -LO https://github.com/oss-review-toolkit/ort/releases/latest/download/ort
          chmod +x ort
          sudo mv ort /usr/local/bin/

   
      - name: Run ORT
        run: |
          ort analyze --input-dir . --output-dir ort-results
          ort report --input-dir ort-results --report-formats StaticHtml


      - name: Build Docker image
        run: |
          docker build -t $DOCKER_IMAGE .

      - name: Push Docker image
        if: github.ref == 'refs/heads/main'
        run: |
          echo "${{ secrets.DOCKER_PASSWORD }}" | docker login -u "${{ secrets.DOCKER_USERNAME }}" --password-stdin
          docker tag $DOCKER_IMAGE myregistry/myapp:latest
          docker push myregistry/myapp:latest
